// GROUPE D’ENTRÉE — CRÉATION DU BUNDLE FINAL PATIENT SUMMARY

//    Ce groupe crée le Bundle final avec Composition et Patient. Ces ressources
//    sont toujours nécessaires dans un document FHIR de type document.

//    Les autres ressources administratives sont créées plus loin, uniquement si le
//    bloc CDA correspondant existe
group TransformPatientSummaryClinicalDocumentToFHIRDocument(source cda : ClinicalDocument, target bundle : Bundle) {
  cda ->  bundle.entry as compositionEntry,  compositionEntry.resource = create('Composition') as composition,  composition.id = uuid(),  compositionEntry.fullUrl = ('urn:uuid:' + %composition.id),  bundle.entry as patientEntry,  patientEntry.resource = create('Patient') as patient,  patient.id = uuid(),  patientEntry.fullUrl = ('urn:uuid:' + %patient.id) then OrchestrateClinicalDocumentFinalTransformation(cda, patient, composition, bundle) "main";
}

// GROUPE D’ORCHESTRATION FINALE — SOCLE FRANÇAIS + MÉTIER PATIENT SUMMARY
//    Ce groupe applique deux traitements successifs :
//    - MapClinicalDocumentHeaderWithFrenchRules, importé depuis l’étape 3, traite
//      l’en-tête CDA, crée les ressources administratives nécessaires, applique le
//      socle générique de l’étape 2 et l’enrichissement français ;
//    - MapPatientSummaryBody traite ensuite les sections Patient Summary et crée
//      les ressources métier à partir des entries CDA.

//    Les ressources optionnelles ne sont donc pas créées au début : elles sont
//    créées uniquement dans les groupes associés aux blocs CDA présents
group OrchestrateClinicalDocumentFinalTransformation(source cda : ClinicalDocument, target patient : Patient, target composition : Composition, target bundle : Bundle) {
  cda then MapClinicalDocumentHeaderWithFrenchRules(cda, bundle, composition, patient) "headerFrenchBase";
  cda then MapPatientSummaryBody(cda, composition, bundle, patient) "patientSummaryBody";
}

// CORPS PATIENT SUMMARY — SECTIONS + ENTRIES MÉTIER
//    Pour chaque section CDA :
//    - MapCDASectionToCompositionSection, importé depuis l’étape 2, crée le
//      squelette commun de la Composition.section avec title, code et text ;
//    - MapPatientSummarySectionEntries crée ensuite les ressources métier selon
//      le code LOINC de la section
group MapPatientSummaryBody(source src : ClinicalDocument, target composition : Composition, target bundle : Bundle, target patient : Patient) {
  src.component as cdaComponent then {
    cdaComponent.structuredBody as structuredBody then {
      structuredBody.component as bodyComponent then {
        bodyComponent.section as section -> composition.section as compSection then {
          section then MapCDASectionToCompositionSection(section, compSection) "sectionBase";
          section then MapPatientSummarySectionEntries(section, compSection, bundle, patient) "patientSummaryEntries";
        };
      } "bodyComponent";
    };
  };
}

// Observation CDA de problème -> Condition FHIR
group PatientSummaryConditionFromProblemObservation(source observation, target condition : Condition, target compSection, target patient : Patient, target bundle : Bundle) {
  observation ->  condition.subject = create('Reference') as subjectRef,  subjectRef.reference = ('urn:uuid:' + %patient.id) "subject";
  observation.id as problemId -> condition.identifier = create('Identifier') as identifier then II(problemId, identifier) "identifier";
  observation.code as problemType -> condition.category = create('CodeableConcept') as category then {
    problemType then CDCodeableConcept(problemType, category) "categoryCoding";
    observation.text as txt then {
      txt.reference as ref then {
        ref.value as value -> category.text = cast(value, 'string') "categoryText";
      };
    };
  } "category";
  observation.value as problemValue -> condition.code = create('CodeableConcept') as code then CDCodeableConcept(problemValue, code) "code";
  observation.effectiveTime as effectiveTime then {
    effectiveTime.low as low -> condition.onset = create('dateTime') as onsetDateTime then TSDateTime(low, onsetDateTime) "onsetDateTime";
    effectiveTime.high as high -> condition.abatement = create('dateTime') as abatementDateTime then TSDateTime(high, abatementDateTime) "abatementDateTime";
  };
  observation.entryRelationship as er then {
    er.observation as severityObs then {
      severityObs.templateId as tid then {
        tid.root as root where root = '1.2.250.1.213.1.1.3.29' then {
          severityObs.value as severityValue -> condition.severity = create('CodeableConcept') as severity then CDCodeableConcept(severityValue, severity) "severity";
        } "severity";
      } "severityTemplate";
    } "severityBlock";
    er.observation as problemStatusObs then {
      problemStatusObs.templateId as tid then {
        tid.root as root where root = '1.2.250.1.213.1.1.3.30' then {
          problemStatusObs.value as statusValue -> condition.clinicalStatus = create('CodeableConcept') as clinicalStatus then {
            statusValue.displayName as display -> clinicalStatus.text = cast(display, 'string') "clinicalStatusText";

              code -> coding.code = cast(code, 'code') "clinicalStatusCode";

              code where code = 'recurrence' -> coding.display = 'Recurrence' "clinicalStatusDisplayRecurrence";
              code where code = 'relapse' -> coding.display = 'Relapse' "clinicalStatusDisplayRelapse";
              code where code = 'inactive' -> coding.display = 'Inactive' "clinicalStatusDisplayInactive";
              code where code = 'remission' -> coding.display = 'Remission' "clinicalStatusDisplayRemission";
              code where code = 'resolved' -> coding.display = 'Resolved' "clinicalStatusDisplayResolved";
            } "clinicalStatusCoding";
          } "clinicalStatus";
        } "clinicalStatus";
      } "clinicalStatusTemplate";
    } "clinicalStatusBlock";
    er.observation as healthStatusObs then {
      healthStatusObs.templateId as tid then {
        tid.root as root where root = '1.2.250.1.213.1.1.3.31' then {
          healthStatusObs.value as healthStatusValue -> condition.stage as stage then {
            healthStatusValue -> stage.summary = create('CodeableConcept') as summary then CDCodeableConcept(healthStatusValue, summary) "stageSummary";
          } "stage";
        } "stageSummary";
      } "stageSummaryTemplate";
    } "stageSummaryBlock";
    er.observation as certaintyObs then {
      certaintyObs.templateId as tid then {
        tid.root as root where root = '1.2.250.1.213.1.1.3.171' then {
          certaintyObs.value as certaintyValue -> condition.verificationStatus = create('CodeableConcept') as verificationStatus then {
            certaintyValue.displayName as display -> verificationStatus.text = cast(display, 'string') "verificationStatusText";
            certaintyValue.code as code -> verificationStatus.coding = create('Coding') as coding then {
              certaintyValue -> coding.system = 'http://terminology.hl7.org/CodeSystem/condition-ver-status' "verificationStatusSystem";
              code -> coding.code = cast(code, 'code') "verificationStatusCode";
              code where code = 'unconfirmed' -> coding.display = 'Unconfirmed' "verificationStatusDisplayUnconfirmed";
              code where code = 'provisional' -> coding.display = 'Provisional' "verificationStatusDisplayProvisional";
              code where code = 'differential' -> coding.display = 'Differential' "verificationStatusDisplayDifferential";
              code where code = 'confirmed' -> coding.display = 'Confirmed' "verificationStatusDisplayConfirmed";
              code where code = 'refuted' -> coding.display = 'Refuted' "verificationStatusDisplayRefuted";
              code where code = 'entered-in-error' -> coding.display = 'Entered in Error' "verificationStatusDisplayEnteredInError";
            } "verificationStatusCoding";
          } "verificationStatus";
        } "verificationStatus";
      } "verificationStatusTemplate";
    } "verificationStatusBlock";
  } "entryRelationships";
  observation.reference as cdaReference then {
    cdaReference.externalDocument as externalDocument ->  bundle.entry as documentReferenceEntry,  documentReferenceEntry.resource = create('DocumentReference') as documentReference,  documentReference.id = uuid(),  documentReferenceEntry.fullUrl = ('urn:uuid:' + %documentReference.id) then {
      externalDocument -> documentReference.status = 'current' "documentReferenceStatus";
      externalDocument ->  documentReference.subject = create('Reference') as subjectRef,  subjectRef.reference = ('urn:uuid:' + %patient.id) "documentReferenceSubject";
      externalDocument.id as externalId -> documentReference.identifier = create('Identifier') as identifier then II(externalId, identifier) "documentReferenceIdentifier";
      externalDocument.text as externalText then {
        externalText.reference as externalRef then {
          externalRef.value as url -> documentReference.content as content then {
            url -> content.attachment = create('Attachment') as attachment then {
              url -> attachment.url = cast(url, 'string') "attachmentUrl";

      } "externalText";
      externalDocument -> condition.evidence as evidence then {
        externalDocument ->  evidence.detail = create('Reference') as detailRef,  detailRef.reference = ('urn:uuid:' + %documentReference.id) "evidenceDetail";
      } "evidence";
    } "externalDocumentReference";
  } "externalReference";
  observation ->  compSection.entry = create('Reference') as sectionRef,  sectionRef.reference = ('urn:uuid:' + %condition.id) "sectionEntry";
}

// Observation CDA d’allergie -> AllergyIntolerance
group AllergyFromCdaObservation(source observation, target allergy : AllergyIntolerance, target compSection, target patient : Patient) {
  observation.id as allergyId -> allergy.identifier = create('Identifier') as identifier then II(allergyId, identifier) "identifier";
  observation ->  allergy.patient = create('Reference') as patientRef,  patientRef.reference = ('urn:uuid:' + %patient.id) "patient";
  observation.participant as participant then {
    participant.participantRole as participantRole then {
      participantRole.playingEntity as playingEntity then {
        playingEntity.code as allergenCode -> allergy.code = create('CodeableConcept') as code then CDCodeableConcept(allergenCode, code) "allergenCode";
      };
    };
  };
  observation.code as allergyTypeCode then {
    allergyTypeCode.code as code where code = '416098002' ->  allergy.type = 'allergy',  allergy.category = 'medication' "medicationAllergyByCode";
    allergyTypeCode.displayName as display where display.lower().contains('intolérance') or display.lower().contains('intolerance') -> allergy.type = 'intolerance' "typeIntolerance";
    allergyTypeCode.displayName as display where (allergyTypeCode.code != '416098002') and (display.lower().contains('allergie') or display.lower().contains('hypersensibilité') or display.lower().contains('hypersensibilite')) and display.lower().contains('intolérance').not() and display.lower().contains('intolerance').not() -> allergy.type = 'allergy' "typeAllergy";
    allergyTypeCode.displayName as display where (allergyTypeCode.code != '416098002') and (display.lower().contains('médicament') or display.lower().contains('medicament') or display.lower().contains('médicamenteuse') or display.lower().contains('medicamenteuse')) -> allergy.category = 'medication' "categoryMedicationByDisplay";
    allergyTypeCode.displayName as display where display.lower().contains('alimentaire') or display.lower().contains('aliment') or display.lower().contains('nourriture') -> allergy.category = 'food' "categoryFood";
    allergyTypeCode.displayName as display where display.lower().contains('environnement') or display.lower().contains('pollen') or display.lower().contains('poussière') or display.lower().contains('poussiere') or display.lower().contains('latex') -> allergy.category = 'environment' "categoryEnvironment";
    allergyTypeCode.displayName as display where display.lower().contains('biologique') or display.lower().contains('vaccin') or display.lower().contains('sérum') or display.lower().contains('serum') -> allergy.category = 'biologic' "categoryBiologic";
  } "typeAndCategory";
  observation.effectiveTime as effectiveTime then {
    effectiveTime.low as low -> allergy.onset = create('dateTime') as onsetDateTime then TSDateTime(low, onsetDateTime) "onsetDateTime";
  };
  observation.entryRelationship as er then {
    er.observation as statusObs then {
      statusObs.templateId as tid then {
        tid.root as root where root = '1.2.250.1.213.1.1.3.30' then {
          statusObs.value as statusValue -> allergy.clinicalStatus = create('CodeableConcept') as clinicalStatus then {
            statusValue.originalText as originalText then {
              originalText.reference as ref then {
                ref.value as value -> clinicalStatus.text = cast(value, 'string') "textFromOriginalTextReference";
              };
              originalText.xmlText as txt where originalText.reference.empty() -> clinicalStatus.text = cast(txt, 'string') "textFromOriginalText";
            };
            statusValue.displayName as display where statusValue.originalText.empty() -> clinicalStatus.text = cast(display, 'string') "textFromDisplayName";
            statusValue.code as code where (code = 'active') or (code = 'actif') -> clinicalStatus.coding = create('Coding') as coding then {
              statusValue -> coding.system = 'http://terminology.hl7.org/CodeSystem/allergyintolerance-clinical' "system";
              statusValue -> coding.code = 'active' "code";
              statusValue -> coding.display = 'Active' "display";
            } "activeFromCode";
            statusValue.code as code where (code = 'inactive') or (code = 'inactif') -> clinicalStatus.coding = create('Coding') as coding then {
              statusValue -> coding.system = 'http://terminology.hl7.org/CodeSystem/allergyintolerance-clinical' "system";
              statusValue -> coding.code = 'inactive' "code";
              statusValue -> coding.display = 'Inactive' "display";
            } "inactiveFromCode";
            statusValue.code as code where (code = 'resolved') or (code = 'resolu') or (code = 'résolu') -> clinicalStatus.coding = create('Coding') as coding then {
              statusValue -> coding.system = 'http://terminology.hl7.org/CodeSystem/allergyintolerance-clinical' "system";
              statusValue -> coding.code = 'resolved' "code";
              statusValue -> coding.display = 'Resolved' "display";
            } "resolvedFromCode";
            statusValue.displayName as display where statusValue.code.empty() and display.lower().contains('actif') and display.lower().contains('inactif').not() -> clinicalStatus.coding = create('Coding') as coding then {
              statusValue -> coding.system = 'http://terminology.hl7.org/CodeSystem/allergyintolerance-clinical' "system";
              statusValue -> coding.code = 'active' "code";
              statusValue -> coding.display = 'Active' "display";
            } "activeFromText";
            statusValue.displayName as display where statusValue.code.empty() and display.lower().contains('inactif') -> clinicalStatus.coding = create('Coding') as coding then {
              statusValue -> coding.system = 'http://terminology.hl7.org/CodeSystem/allergyintolerance-clinical' "system";
              statusValue -> coding.code = 'inactive' "code";
              statusValue -> coding.display = 'Inactive' "display";
            } "inactiveFromText";
            statusValue.displayName as display where statusValue.code.empty() and (display.lower().contains('résolu') or display.lower().contains('resolu')) -> clinicalStatus.coding = create('Coding') as coding then {
              statusValue -> coding.system = 'http://terminology.hl7.org/CodeSystem/allergyintolerance-clinical' "system";
              statusValue -> coding.code = 'resolved' "code";
              statusValue -> coding.display = 'Resolved' "display";
            } "resolvedFromText";
            statusValue where statusValue.code.empty() and statusValue.displayName.empty() and statusValue.originalText.empty() -> clinicalStatus.extension = create('Extension') as absentReason then {
              statusValue -> absentReason.url = 'http://hl7.org/fhir/StructureDefinition/data-absent-reason' "url";
              statusValue -> absentReason.value = cast('unknown', 'code') "valueCode";
            } "clinicalStatusAbsentReason";
          } "clinicalStatus";
        } "clinicalStatusBlock";
      } "statusTemplate";
    } "statusBlock";
    er.observation as statusObs then {
      statusObs.templateId as tid then {
        tid.root as root where root = '1.2.250.1.213.1.1.3.30' then {
          statusObs where statusObs.value.empty() -> allergy.clinicalStatus = create('CodeableConcept') as clinicalStatus then {
            statusObs -> clinicalStatus.extension = create('Extension') as absentReason then {
              statusObs -> absentReason.url = 'http://hl7.org/fhir/StructureDefinition/data-absent-reason' "url";
              statusObs -> absentReason.value = cast('unknown', 'code') "valueCode";
            } "dataAbsentReason";
          } "clinicalStatusNoValue";

// Statut CDA -> Observation.status
group ObservationStatusFromCdaStatus(source statusCode, target fhirObservation : Observation) {
  statusCode.code as code where code = 'completed' -> fhirObservation.status = 'final' "completed";
  statusCode.code as code where code = 'active' -> fhirObservation.status = 'registered' "active";
  statusCode.code as code where code = 'aborted' -> fhirObservation.status = 'cancelled' "aborted";
  statusCode.code as code where code = 'cancelled' -> fhirObservation.status = 'cancelled' "cancelled";
}

// effectiveTime CDA -> Observation.effective[x]
group ObservationEffectiveFromEffectiveTime(source effectiveTime, target fhirObservation : Observation) {
  effectiveTime.value as value -> fhirObservation.effective = create('dateTime') as effectiveDateTime then TSDateTime(effectiveTime, effectiveDateTime) "effectiveDateTime";
  effectiveTime.low as low where effectiveTime.high.empty() -> fhirObservation.effective = create('dateTime') as effectiveDateTime then TSDateTime(low, effectiveDateTime) "effectiveLowAsDateTime";
  effectiveTime.low as low where effectiveTime.high.exists() -> fhirObservation.effective = create('Period') as effectivePeriod then {
    low -> effectivePeriod.start = create('dateTime') as start then TSDateTime(low, start) "start";
    effectiveTime.high as high -> effectivePeriod.end = create('dateTime') as end then TSDateTime(high, end) "end";
  } "effectivePeriod";
  effectiveTime where effectiveTime.value.empty() and effectiveTime.low.empty() and effectiveTime.high.empty() -> fhirObservation.effective = create('dateTime') as effectiveDateTime then {
    effectiveTime -> effectiveDateTime.extension = create('Extension') as absentReason then {
      effectiveTime -> absentReason.url = 'http://hl7.org/fhir/StructureDefinition/data-absent-reason' "url";
      effectiveTime -> absentReason.value = cast('unknown', 'code') "valueCode";
    } "dataAbsentReason";
  } "effectiveTimePresentWithoutValue";
}

// Valeur CDA -> Observation.value[x]

  value.unit as unit -> fhirObservation.value = create('Quantity') as quantity then {

  } "valueQuantity";
  value.code as code -> fhirObservation.value = create('CodeableConcept') as valueCodeableConcept then CDCodeableConcept(value, valueCodeableConcept) "valueCodeableConcept";
  value.originalText as originalText where value.code.empty() and value.unit.empty() then {
    originalText.reference as reference then {
      reference.value as text ->  fhirObservation.value = create('CodeableConcept') as valueCodeableConcept,  valueCodeableConcept.text = cast(text, 'string') "valueCodeableConceptText";
    };
  } "originalTextOnly";
  value.value as v where value.unit.empty() and value.code.empty() and (v = 'true') -> fhirObservation.value = cast(v, 'boolean') "valueBooleanTrue";

}
