// GROUPE D’ENTRÉE — INITIALISATION DU DOCUMENT FHIR À PARTIR DU CDA

//    À partir de l’analyse du ClinicalDocument CDA, ce groupe initialise les
//    ressources FHIR nécessaires pour représenter les éléments CDA principaux :
//    Composition et Patient.

//    Ces ressources sont créées vides avec un uuid(), ajoutées au Bundle, puis
//    transmises au groupe d’orchestration.

//    Les autres ressources seront créées ultérieurement, dans les groupes
//    correspondant aux blocs CDA traités.
group TransformClinicalDocumentToFHIRDocument(source cda : ClinicalDocument, target bundle : Bundle) {
  cda ->  bundle.entry as compositionEntry,  compositionEntry.resource = create('Composition') as composition,  composition.id = uuid(),  compositionEntry.fullUrl = ('urn:uuid:' + %composition.id),  bundle.entry as patientEntry,  patientEntry.resource = create('Patient') as patient,  patient.id = uuid(),  patientEntry.fullUrl = ('urn:uuid:' + %patient.id) then OrchestrateClinicalDocumentTransformation(cda, patient, composition, bundle) "main";
}

// GROUPE D’ORCHESTRATION — TRAITEMENT DU CLINICALDOCUMENT CDA

//    Ce groupe permet de traiter le ClinicalDocument CDA en deux parties :
//    - l’en-tête CDA via l'appel du groupe MapClinicalDocumentHeader ;
//    - le corps CDA via l'appelle du groupe MapClinicalDocumentBody.

//    Les ressources initialisées dans le groupe d’entrée, Bundle, Composition et
//    Patient, servent de support de sortie pour le traitement CDA
group OrchestrateClinicalDocumentTransformation(source cda : ClinicalDocument, target patient : Patient, target composition : Composition, target bundle : Bundle) {
  cda then MapClinicalDocumentHeader(cda, bundle, composition, patient) "header";
  cda then MapClinicalDocumentBody(cda, composition, bundle, patient) "body";
}

// GROUPE — TRAITEMENT DE L’EN-TÊTE DU CLINICALDOCUMENT CDA

//    Ce groupe traite l’en-tête du ClinicalDocument CDA bloc par bloc.

//    Pour chaque élément de l’en-tête CDA, il appelle le groupe spécialisé
//    correspondant :
//    - id, setId, versionNumber et effectiveTime sont transmis à
//      MapHeaderDocumentIdentity pour alimenter les métadonnées du Bundle et de
//      la Composition ;
//    - languageCode, code, title, effectiveTime et confidentialityCode sont transmis
//      à MapHeaderDocumentDescription pour alimenter la description documentaire ;
//    - recordTarget.patientRole est transmis à MapHeaderRecordTarget pour alimenter
//      le Patient et la référence Composition.subject ;
//    - custodian est transmis à MapHeaderCustodian pour créer et alimenter
//      l’Organization responsable de la conservation du document ;
//    - author est transmis à MapHeaderAuthor pour créer et alimenter le Practitioner,
//      l’Organization et le PractitionerRole de l’auteur ;
//    - legalAuthenticator est transmis à MapHeaderLegalAuthenticator pour créer et
//      alimenter le signataire légal ;
//    - documentationOf.serviceEvent est transmis à MapHeaderDocumentationOf pour
//      alimenter Composition.event ;
//    - componentOf.encompassingEncounter est transmis à MapHeaderEncounterContext
//      pour créer et alimenter Encounter, Location et Organization si le bloc CDA
//      est présent ;
//    - informant.relatedEntity classCode=ECON est transmis à MapHeaderEmergencyContact
//      pour créer la RelatedPerson correspondant à la personne à prévenir ;
//    - informant.relatedEntity classCode=NOK est transmis à MapHeaderTrustedPerson
//      pour créer la RelatedPerson correspondant à la personne de confiance ;
//    - recordTarget.patientRole.patient.guardian est transmis à MapHeaderGuardian
//      pour créer la RelatedPerson correspondant au guardian ;
//    - relatedDocument est transmis à MapHeaderRelatedDocument pour alimenter
//      Composition.relatesTo.

//    Les ressources optionnelles sont créées dans le groupe qui traite le bloc CDA
//    correspondant. Ainsi, une ressource FHIR n’est créée que lorsqu’un élément CDA
//    justifie son existence
group MapClinicalDocumentHeader(source cda : ClinicalDocument, target bundle : Bundle, target composition : Composition, target patient : Patient) {
  cda then MapHeaderDocumentIdentity(cda, bundle, composition) "documentIdentity";
  cda then MapHeaderDocumentDescription(cda, composition, patient) "documentDescription";
  cda then MapHeaderRecordTarget(cda, patient, composition) "recordTarget";
  cda then MapHeaderCustodian(cda, composition, bundle) "custodian";
  cda then MapHeaderAuthor(cda, composition, bundle) "author";
  cda then MapHeaderLegalAuthenticator(cda, composition, bundle) "legalAuthenticator";
  cda then MapHeaderDocumentationOf(cda, composition) "documentationOf";
  cda then MapHeaderEncounterContext(cda, composition, bundle, patient) "encounterContext";
  cda then MapHeaderEmergencyContact(cda, bundle, patient) "emergencyContact";
  cda then MapHeaderTrustedPerson(cda, bundle, patient) "trustedPerson";
  cda then MapHeaderGuardian(cda, bundle, patient) "guardian";
  cda then MapHeaderRelatedDocument(cda, composition) "relatedDocument";
}

// groupe qui traite l'identité du document CDA -> métadonnées Bundle / Composition FHIR
group MapHeaderDocumentIdentity(source src : ClinicalDocument, target bundle : Bundle, target composition : Composition) {
  src -> bundle.id = uuid() "bundleId";
  src -> bundle.type = 'document' "bundleType";
  src -> bundle.meta as bundleMeta then {
    src.versionNumber as versionNumber then {
      versionNumber.value as version -> bundleMeta.versionId = cast(version, 'string') "bundleMetaVersionId";
    } "bundleVersionNumber";
    src.effectiveTime as effectiveTime -> bundleMeta.lastUpdated = create('instant') as lastUpdated then TSInstant(effectiveTime, lastUpdated) "bundleMetaLastUpdated";
  } "bundleMeta";
  src.id as cdaId -> bundle.identifier = create('Identifier') as identifier then {
    cdaId.root as root where cdaId.extension.exists() -> identifier.system = append('urn:oid:', root) "bundleIdentifierSystemWithExtension";
    cdaId.extension as extension -> identifier.value = extension "bundleIdentifierValueWithExtension";
    cdaId.root as root where cdaId.extension.empty() -> identifier.system = 'urn:ietf:rfc:3986' "bundleIdentifierSystemOid";
    cdaId.root as root where cdaId.extension.empty() -> identifier.value = append('urn:oid:', root) "bundleIdentifierValueOid";
  } "bundleIdentifier";
  src.effectiveTime as effectiveTime -> bundle.timestamp = create('instant') as timestamp then TSInstant(effectiveTime, timestamp) "bundleTimestamp";
  src.setId as setId -> composition.identifier = create('Identifier') as identifier then II(setId, identifier) "compositionSetIdentifier";
  src.id as id where src.setId.empty() -> composition.identifier = create('Identifier') as identifier then II(id, identifier) "compositionIdentifier";
  src -> composition.status = 'final' "compositionStatus";
  src -> composition.meta as compositionMeta then {
    src.versionNumber as versionNumber then {
      versionNumber.value as version -> compositionMeta.versionId = cast(version, 'string') "compositionMetaVersionId";
    } "compositionVersionNumber";
    src.effectiveTime as effectiveTime -> compositionMeta.lastUpdated = create('instant') as lastUpdated then TSInstant(effectiveTime, lastUpdated) "compositionMetaLastUpdated";
  } "compositionMeta";
}

// groupe qui traite la description du document CDA -> FHIR Composition
group MapHeaderDocumentDescription(source src : ClinicalDocument, target composition : Composition, target patient : Patient) {
  src.languageCode as languageCode -> composition.language = create('code') as code then CSCode(languageCode, code) "language";
  src.code as srcCode -> composition.type = create('CodeableConcept') as cc then CDCodeableConcept(srcCode, cc) "type";
  src.title as title -> composition.title = (title.xmlText);
  src ->  composition.subject = create('Reference') as subjectRef,  subjectRef.reference = ('urn:uuid:' + %patient.id) "subject";
  src.effectiveTime as effectiveTime -> composition.date = create('dateTime') as date then TSDateTime(effectiveTime, date) "date";
  src.confidentialityCode as confCode -> composition.confidentiality = create('code') as code then CSCode(confCode, code) "confidentiality";
}

// groupe qui traite: CDA recordTarget.patientRole -> FHIR Patient
group MapHeaderRecordTarget(source src : ClinicalDocument, target patient : Patient, target composition : Composition) {
  src.recordTarget as recordTarget then {
    recordTarget.patientRole as patientRole then {
      patientRole.addr as srcAddr -> patient.address = create('Address') as address then ADAddress(srcAddr, address) "address";
      patientRole.telecom as srcTelecom -> patient.telecom = create('ContactPoint') as contactPoint then TELContactPoint(srcTelecom, contactPoint);
      patientRole.patient as patientCda then {
        patientCda.administrativeGenderCode as gender then {
          gender.code as v -> patient.gender = translate(v, 'https://interop.esante.gouv.fr/ig/fhir/mappingcdafhir/ConceptMap/cm-v3-administrative-gender', 'code') "gender";
        } "gender";
        patientCda.birthTime as birthTime -> patient.birthDate = create('date') as date then TSDate(birthTime, date) "birthDate";
        patientCda.deceasedInd as indicator where patientCda.deceasedTime.empty() -> patient.deceased = create('boolean') as bool then {
          indicator.value as v -> bool.value = v;
        } "deceasedBoolean";
        patientCda.deceasedTime as dTime -> patient.deceased = create('dateTime') as dt then TSDateTime(dTime, dt) "deceasedDateTime";
      } "patientData";
    };
  };
}

// groupe qui traite custodian  (l'organisation)
// L’Organization est créée uniquement si le bloc custodian existe dans le CDA.
group MapHeaderCustodian(source src : ClinicalDocument, target composition : Composition, target bundle : Bundle) {
  src.custodian as custodian then {
    custodian.assignedCustodian as assignedCustodian then {
      assignedCustodian.representedCustodianOrganization as representedOrg ->  bundle.entry as custodianOrgEntry,  custodianOrgEntry.resource = create('Organization') as custodianOrganization,  custodianOrganization.id = uuid(),  custodianOrgEntry.fullUrl = ('urn:uuid:' + %custodianOrganization.id) then {
        representedOrg then MapCDAOrganizationToOrganization(representedOrg, custodianOrganization) "custodianOrganization";
        representedOrg ->  composition.custodian = create('Reference') as custodianRef,  custodianRef.reference = ('urn:uuid:' + %custodianOrganization.id) "custodianRef";
      } "custodianOrganizationResource";
    };
  };
}

// GROUPE — TRAITEMENT DE L’AUTEUR DU DOCUMENT CDA

//    Ce groupe traite le bloc author de l’en-tête CDA.

//    Si le bloc author existe, il crée un Practitioner pour représenter l’auteur.
//    Si author.assignedAuthor.representedOrganization est présent, il crée aussi
//    une Organization et un PractitionerRole afin de représenter l’auteur dans son
//    organisation.

//    La Composition référence ensuite le PractitionerRole si l’organisation existe,
//    sinon elle référence directement le Practitioner.
group MapHeaderAuthor(source src : ClinicalDocument, target composition : Composition, target bundle : Bundle) {
  src.author as author then {
    author.assignedAuthor as assignedAuthor ->  bundle.entry as practitionerEntry,  practitionerEntry.resource = create('Practitioner') as authorPractitioner,  authorPractitioner.id = uuid(),  practitionerEntry.fullUrl = ('urn:uuid:' + %authorPractitioner.id) then {
      assignedAuthor then MapCDAAssignedEntityToPractitioner(assignedAuthor, authorPractitioner) "authorPractitioner";
      assignedAuthor.representedOrganization as representedOrganization ->  bundle.entry as organizationEntry,  organizationEntry.resource = create('Organization') as authorOrganization,  authorOrganization.id = uuid(),  organizationEntry.fullUrl = ('urn:uuid:' + %authorOrganization.id),  bundle.entry as roleEntry,  roleEntry.resource = create('PractitionerRole') as authorRole,  authorRole.id = uuid(),  roleEntry.fullUrl = ('urn:uuid:' + %authorRole.id) then {
        representedOrganization then MapCDAOrganizationToOrganization(representedOrganization, authorOrganization) "authorOrganization";
        assignedAuthor then MapCDAAssignedEntityToPractitionerRole(assignedAuthor, authorRole, authorPractitioner, authorOrganization) "authorRole";
        assignedAuthor ->  composition.author as authorRef,  authorRef.reference = ('urn:uuid:' + %authorRole.id) "compositionAuthorRoleRef";
      } "authorWithOrganization";
      assignedAuthor where assignedAuthor.representedOrganization.empty() ->  composition.author as authorRef,  authorRef.reference = ('urn:uuid:' + %authorPractitioner.id) "compositionAuthorPractitionerRef";
    } "authorPractitionerResource";
  };
}

// GROUPE — TRAITEMENT DU SIGNATAIRE LÉGAL CDA
//    Ce groupe traite le bloc legalAuthenticator de l’en-tête CDA

//    Si le bloc legalAuthenticator existe, il crée un Practitioner pour représenter
//    le signataire légal. Si legalAuthenticator.assignedEntity.representedOrganization
//    est présent, il crée aussi une Organization et un PractitionerRole.

//    La Composition est alimentée via Composition.attester. La partie attester.party
//    référence le PractitionerRole si l’organisation existe, sinon elle référence
//    directement le Practitioner.
group MapHeaderLegalAuthenticator(source src : ClinicalDocument, target composition : Composition, target bundle : Bundle) {
  src.legalAuthenticator as legalAuthenticator then {
    legalAuthenticator.assignedEntity as assignedEntity ->  bundle.entry as practitionerEntry,  practitionerEntry.resource = create('Practitioner') as legalPractitioner,  legalPractitioner.id = uuid(),  practitionerEntry.fullUrl = ('urn:uuid:' + %legalPractitioner.id) then {
      assignedEntity then MapCDAAssignedEntityToPractitioner(assignedEntity, legalPractitioner) "legalPractitioner";
      legalAuthenticator -> composition.attester as attester then {
        legalAuthenticator -> attester.mode = 'legal' "mode";
        legalAuthenticator.time as time -> attester.time = create('dateTime') as dt then TSDateTime(time, dt);
        assignedEntity.representedOrganization as representedOrganization ->  bundle.entry as organizationEntry,  organizationEntry.resource = create('Organization') as legalOrganization,  legalOrganization.id = uuid(),  organizationEntry.fullUrl = ('urn:uuid:' + %legalOrganization.id),  bundle.entry as roleEntry,  roleEntry.resource = create('PractitionerRole') as legalRole,  legalRole.id = uuid(),  roleEntry.fullUrl = ('urn:uuid:' + %legalRole.id) then {
          representedOrganization then MapCDAOrganizationToOrganization(representedOrganization, legalOrganization) "legalOrganization";
          assignedEntity then MapCDAAssignedEntityToPractitionerRole(assignedEntity, legalRole, legalPractitioner, legalOrganization) "legalRole";
          assignedEntity ->  attester.party = create('Reference') as partyRef,  partyRef.reference = ('urn:uuid:' + %legalRole.id) "partyRole";
        } "legalWithOrganization";
        assignedEntity where assignedEntity.representedOrganization.empty() ->  attester.party = create('Reference') as partyRef,  partyRef.reference = ('urn:uuid:' + %legalPractitioner.id) "partyPractitioner";
      } "attester";
    } "legalPractitionerResource";
  };
}

// GROUPE — TRAITEMENT DE L’ÉVÉNEMENT DOCUMENTÉ CDA
group MapHeaderDocumentationOf(source src : ClinicalDocument, target composition : Composition) {
  src.documentationOf as documentationOf then {
    documentationOf.serviceEvent as serviceEvent -> composition.event as event then {
      serviceEvent.code as eventCode -> event.code = create('CodeableConcept') as cc then CDCodeableConcept(eventCode, cc) "eventCode";
      serviceEvent.effectiveTime as eff -> event.period = create('Period') as period then IVLTSPeriod(eff, period) "eventPeriod";
    } "event";
  };
}

// GROUPE — TRAITEMENT DU CONTEXTE DE PRISE EN CHARGE CDA

//    Ce groupe traite le bloc componentOf.encompassingEncounter de l’en-tête CDA.

//    Si encompassingEncounter existe, il crée une ressource Encounter et l’associe
//    à la Composition.

//    Le remplissage générique de l’Encounter est délégué au groupe commun :
//    MapCDAEncompassingEncounterToEncounter.

//    Si encompassingEncounter.location.healthCareFacility est également présent,
//    il crée une Location et une Organization pour représenter le lieu et la
//    structure de prise en charge, puis appelle :
//    MapCDAHealthCareFacilityToLocationContext.
group MapHeaderEncounterContext(source src : ClinicalDocument, target composition : Composition, target bundle : Bundle, target patient : Patient) {
  src.componentOf as componentOf then {
    componentOf.encompassingEncounter as encompassingEncounter ->  bundle.entry as encounterEntry,  encounterEntry.resource = create('Encounter') as encounter,  encounter.id = uuid(),  encounterEntry.fullUrl = ('urn:uuid:' + %encounter.id) then {
      encompassingEncounter ->  composition.encounter = create('Reference') as encRef,  encRef.reference = ('urn:uuid:' + %encounter.id) "compositionEncounterRef";
      encompassingEncounter then MapCDAEncompassingEncounterToEncounter(encompassingEncounter, encounter, patient) "encounterBase";
      encompassingEncounter.location as srcLocation then {
        srcLocation.healthCareFacility as facility ->  bundle.entry as locationEntry,  locationEntry.resource = create('Location') as location,  location.id = uuid(),  locationEntry.fullUrl = ('urn:uuid:' + %location.id),  bundle.entry as facilityOrganizationEntry,  facilityOrganizationEntry.resource = create('Organization') as facilityOrganization,  facilityOrganization.id = uuid(),  facilityOrganizationEntry.fullUrl = ('urn:uuid:' + %facilityOrganization.id) then MapCDAHealthCareFacilityToLocationContext(facility, encounter, location, facilityOrganization) "facilityContext";
      };
    } "encounterResource";
  };
}

// GROUPE COMMUN — ENCOMPASSINGENCOUNTER CDA VERS ENCOUNTER FHIR

//    Ce groupe transforme les informations génériques du bloc CDA
//    componentOf.encompassingEncounter vers une ressource FHIR Encounter déjà créée
//    par le groupe appelant.

//    Il alimente :
//    - Encounter.identifier ;
//    - Encounter.subject ;
//    - Encounter.status ;
//    - Encounter.class ;
//    - Encounter.period.

//    Si le code CDA de l’encompassingEncounter est absent, Encounter.class est
//    créé avec l’extension data-absent-reason afin de respecter l’obligation FHIR
//    sans inventer une information absente du CDA
group MapCDAEncompassingEncounterToEncounter(source encompassingEncounter : EncompassingEncounter, target encounter : Encounter, target patient : Patient) {
  encompassingEncounter.id as id -> encounter.identifier = create('Identifier') as identifier then {
    id -> identifier then II(id, identifier) "baseIdentifier";
    id -> identifier.type = create('CodeableConcept') as idType then {
      id -> idType.coding = create('Coding') as coding then {
        id -> coding.system = 'http://terminology.hl7.org/CodeSystem/v2-0203' "system";
        id -> coding.code = 'VN' "code";
        id -> coding.display = 'Visit number' "display";
      } "coding";
    } "type";
  } "identifier";
  encompassingEncounter ->  encounter.subject = create('Reference') as subjectRef,  subjectRef.reference = ('urn:uuid:' + %patient.id) "subject";
  encompassingEncounter -> encounter.status = 'finished' "status";
  encompassingEncounter.code as srcCode -> encounter.class = create('Coding') as coding then CDCoding(srcCode, coding) "class";
  encompassingEncounter where encompassingEncounter.code.empty() -> encounter.class = create('Coding') as coding then {
    encompassingEncounter -> coding.extension = create('Extension') as absentReason then {
      encompassingEncounter -> absentReason.url = 'http://hl7.org/fhir/StructureDefinition/data-absent-reason' "url";
      encompassingEncounter -> absentReason.value = cast('unknown', 'code') "valueCode";
    } "dataAbsentReason";
  } "classAbsentReason";
  encompassingEncounter.effectiveTime as effTime where effTime.value.exists() or effTime.low.exists() or effTime.high.exists() -> encounter.period = create('Period') as period then IVLTSPeriod(effTime, period) "period";
}

// GROUPE — TRAITEMENT DE LA PERSONNE À PRÉVENIR CDA

//    Ce groupe traite les blocs informant.relatedEntity dont classCode = ECON.

//    Si un informant ECON est présent dans le CDA, il crée une RelatedPerson pour
//    représenter la personne à prévenir en cas d’urgence, puis l’associe au Patient.

//    La ressource RelatedPerson est donc créée uniquement si ce bloc CDA existe
group MapHeaderEmergencyContact(source src : ClinicalDocument, target bundle : Bundle, target patient : Patient) {
  src.informant as informant then {
    informant.relatedEntity as relatedEntity where relatedEntity.classCode = 'ECON' ->  bundle.entry as relatedPersonEntry,  relatedPersonEntry.resource = create('RelatedPerson') as emergencyContactRelatedPerson,  emergencyContactRelatedPerson.id = uuid(),  relatedPersonEntry.fullUrl = ('urn:uuid:' + %emergencyContactRelatedPerson.id) then MapCDARelatedEntityToRelatedPerson(relatedEntity, emergencyContactRelatedPerson, patient) "relatedPersonECON";
  };
}

// GROUPE — TRAITEMENT DE LA PERSONNE DE CONFIANCE CDA

//    Ce groupe traite les blocs informant.relatedEntity dont classCode = NOK.

//    Si un informant NOK est présent dans le CDA, il crée une RelatedPerson pour
//    représenter la personne de confiance, puis l’associe au Patient.

//    La ressource RelatedPerson est donc créée uniquement si ce bloc CDA existe.
group MapHeaderTrustedPerson(source src : ClinicalDocument, target bundle : Bundle, target patient : Patient) {
  src.informant as informant then {
    informant.relatedEntity as relatedEntity where relatedEntity.classCode = 'NOK' ->  bundle.entry as relatedPersonEntry,  relatedPersonEntry.resource = create('RelatedPerson') as trustedPersonRelatedPerson,  trustedPersonRelatedPerson.id = uuid(),  relatedPersonEntry.fullUrl = ('urn:uuid:' + %trustedPersonRelatedPerson.id) then MapCDARelatedEntityToRelatedPerson(relatedEntity, trustedPersonRelatedPerson, patient) "relatedPersonNOK";
  };
}

// GROUPE — TRAITEMENT DU GUARDIAN CDA

//    Ce groupe traite le bloc recordTarget.patientRole.patient.guardian.

//    Si un guardian est présent dans le CDA, il crée une RelatedPerson pour
//    représenter le représentant légal ou responsable du patient.

//    La ressource RelatedPerson est donc créée uniquement si le bloc guardian existe
group MapHeaderGuardian(source src : ClinicalDocument, target bundle : Bundle, target patient : Patient) {
  src.recordTarget as recordTarget then {
    recordTarget.patientRole as patientRole then {
      patientRole.patient as patientCda then {
        patientCda.guardian as guardian ->  bundle.entry as guardianEntry,  guardianEntry.resource = create('RelatedPerson') as guardianRelatedPerson,  guardianRelatedPerson.id = uuid(),  guardianEntry.fullUrl = ('urn:uuid:' + %guardianRelatedPerson.id) then {
          guardian ->  guardianRelatedPerson.patient = create('Reference') as patientRef,  patientRef.reference = ('urn:uuid:' + %patient.id) "patientRef";
          guardian.id as id -> guardianRelatedPerson.identifier = create('Identifier') as identifier then II(id, identifier) "identifier";
          guardian.addr as addr -> guardianRelatedPerson.address = create('Address') as address then ADAddress(addr, address) "address";
          guardian.telecom as telecom -> guardianRelatedPerson.telecom = create('ContactPoint') as contactPoint then TELContactPoint(telecom, contactPoint);
          guardian.guardianPerson as guardianPerson then {
            guardianPerson.name as name -> guardianRelatedPerson.name = create('HumanName') as humanName then ENHumanName(name, humanName);
          };
        } "guardianResource";
      };
    };
  };
}

// GROUPE — TRAITEMENT DU DOCUMENT LIÉ CDA

//    Ce groupe traite le bloc relatedDocument du CDA.

//    Si relatedDocument est présent, il alimente Composition.relatesTo afin de
//    représenter le lien avec un document antérieur, par exemple un document
//    remplacé.

//    Ce groupe ne crée pas de ressource FHIR indépendante. Il complète uniquement
//    la Composition déjà initialisée
group MapHeaderRelatedDocument(source src : ClinicalDocument, target composition : Composition) {
  src.relatedDocument as relatedDoc -> composition.relatesTo as relatesTo then {
    relatedDoc.typeCode as typeCode where typeCode = 'RPLC' -> relatesTo.code = 'replaces' "replaces";
    relatedDoc.parentDocument as parentDoc then {
      parentDoc.id as id -> relatesTo.target = create('Identifier') as identifier then II(id, identifier) "targetIdentifier";
    };
  };
}

// GROUPE COMMUN — ORGANISATION CDA VERS ORGANIZATION FHIR

//    Ce groupe transforme un élément CDA de type organisation vers une ressource
//    FHIR Organization déjà créée par le groupe appelant.

//    Il alimente les informations génériques de l’organisation :
//    identifier, name, telecom et address
group MapCDAOrganizationToOrganization(source src, target organization : Organization) {
  src.id as id -> organization.identifier = create('Identifier') as identifier then II(id, identifier) "identifier";
  src.name as orgName then {
    orgName.item as item -> organization.name = (item.xmlText) "name";
  } "orgName";
  src.telecom as srcTelecom -> organization.telecom = create('ContactPoint') as contactPoint then TELContactPoint(srcTelecom, contactPoint);
  src.addr as srcAddr -> organization.address = create('Address') as address then ADAddress(srcAddr, address) "address";
}

// GROUPE COMMUN — ASSIGNEDENTITY CDA VERS PRACTITIONER FHIR

//    Ce groupe transforme un élément CDA de type assignedEntity vers une ressource
//    FHIR Practitioner déjà créée par le groupe appelant.

//    Il alimente les informations génériques du professionnel :
//    identifier, address, telecom et name
group MapCDAAssignedEntityToPractitioner(source src, target practitioner : Practitioner) {
  src.id as id -> practitioner.identifier = create('Identifier') as identifier then II(id, identifier) "identifier";
  src.addr as addr -> practitioner.address = create('Address') as address then ADAddress(addr, address) "address";
  src.telecom as telecom -> practitioner.telecom = create('ContactPoint') as contactPoint then TELContactPoint(telecom, contactPoint);
  src.assignedPerson as person then {
    person.name as pName -> practitioner.name = create('HumanName') as humanName then ENHumanName(pName, humanName) "assignedName";
  };
}

// GROUPE COMMUN — ASSIGNEDENTITY CDA VERS PRACTITIONERROLE FHIR

//    Ce groupe transforme le lien fonctionnel porté par un assignedEntity CDA vers
//    une ressource FHIR PractitionerRole déjà créée par le groupe appelant.

//    Il alimente les références PractitionerRole.practitioner et
//    PractitionerRole.organization à partir des ressources Practitioner et
//    Organization déjà créées
group MapCDAAssignedEntityToPractitionerRole(source src, target role : PractitionerRole, target practitioner : Practitioner, target organization : Organization) {
  src ->  role.practitioner = create('Reference') as refPractitioner,  refPractitioner.reference = ('urn:uuid:' + %practitioner.id) "practitionerRef";
  src ->  role.organization = create('Reference') as refOrganization,  refOrganization.reference = ('urn:uuid:' + %organization.id) "organizationRef";
}

// GROUPE COMMUN — RELATEDENTITY CDA VERS RELATEDPERSON FHIR

//    Ce groupe transforme un relatedEntity CDA vers une ressource FHIR RelatedPerson
//    déjà créée par le groupe appelant.

//    Il alimente le lien avec le Patient ainsi que les informations génériques :
//    identifier, address, telecom et name.

//    Le champ relationship n’est pas alimenté dans cette étape afin d’éviter un
//    double mapping. Il est enrichi ensuite dans l’étape française avec le système
//    NOS approprié
group MapCDARelatedEntityToRelatedPerson(source relatedEntity : RelatedEntity, target relatedPerson : RelatedPerson, target patient : Patient) {
  relatedEntity ->  relatedPerson.patient = create('Reference') as patientRef,  patientRef.reference = ('urn:uuid:' + %patient.id) "patientRef";
  relatedEntity.id as id -> relatedPerson.identifier = create('Identifier') as identifier then II(id, identifier) "identifier";
  relatedEntity.addr as addr -> relatedPerson.address = create('Address') as address then ADAddress(addr, address) "address";
  relatedEntity.telecom as telecom -> relatedPerson.telecom = create('ContactPoint') as contactPoint then TELContactPoint(telecom, contactPoint);
  relatedEntity.relatedPerson as person then {
    person.name as name -> relatedPerson.name = create('HumanName') as humanName then ENHumanName(name, humanName);
  };
}

// GROUPE COMMUN — HEALTHCAREFACILITY CDA VERS CONTEXTE DE LIEU FHIR

//    Ce groupe traite le bloc healthCareFacility du CDA.

//    Il complète les ressources Encounter, Location et Organization déjà créées
//    par le groupe appelant lorsque le bloc healthCareFacility existe.

//    Il alimente notamment :
//    - Encounter.location ;
//    - Encounter.serviceProvider ;
//    - Location.managingOrganization ;
//    - Location.name ;
//    - Location.type ;
//    - Organization représentant la structure de prise en charge.
group MapCDAHealthCareFacilityToLocationContext(source facility : HealthCareFacility, target encounter : Encounter, target location : Location, target facilityOrganization : Organization) {
  facility -> encounter.location as encLocation then {
    facility ->  encLocation.location = create('Reference') as locRef,  locRef.reference = ('urn:uuid:' + %location.id) "locationRef";
  } "encounterLocation";
  facility ->  encounter.serviceProvider = create('Reference') as orgRef,  orgRef.reference = ('urn:uuid:' + %facilityOrganization.id) "serviceProvider";
  facility ->  location.managingOrganization = create('Reference') as managingOrgRef,  managingOrgRef.reference = ('urn:uuid:' + %facilityOrganization.id) "managingOrganization";
  facility.location as facilityLocation then {
    facilityLocation.name as locName then {
      locName.item as item -> location.name = (item.xmlText) "locationName";
    } "locationName";
  } "facilityLocation";
  facility.code as facilityCode -> location.type = create('CodeableConcept') as cc then CDCodeableConcept(facilityCode, cc) "locationType";
  facility.serviceProviderOrganization as serviceProviderOrganization then {
    serviceProviderOrganization then MapCDAOrganizationToOrganization(serviceProviderOrganization, facilityOrganization) "facilityOrganization";
  };
}

// CORPS CDA -> SECTIONS FHIR

//    Cette partie transforme uniquement le squelette du structuredBody CDA.
//    Les entries métier ne sont pas transformées dans l’étape 2
// CDA component.structuredBody -> FHIR Composition.section
group MapClinicalDocumentBody(source src : ClinicalDocument, target composition : Composition, target bundle : Bundle, target patient : Patient) {
  src.component as cdaComponent then {
    cdaComponent.structuredBody as structuredBody then {
      structuredBody.component as bodyComponent then {
        bodyComponent.section as section -> composition.section as compSection then {
          section then MapCDASectionToCompositionSection(section, compSection) "sectionBase";
        };
      } "bodyComponent";
    };
  };
}

// CDA section -> FHIR Composition.section title/code/text
group MapCDASectionToCompositionSection(source section, target compSection) {
  section.title as sectionTitle -> compSection.title = (sectionTitle.xmlText) "sectionTitle";
  section.code as sectionCode -> compSection.code = create('CodeableConcept') as cc then CDCodeableConcept(sectionCode, cc) "sectionCode";
  section.text as sectionText -> compSection.text = create('Narrative') as narrative then {
    sectionText -> narrative.status = 'generated' "narrativeStatus";
    sectionText -> narrative.div = cast(sectionText, 'string') "narrativeDiv";
  } "sectionText";
}
