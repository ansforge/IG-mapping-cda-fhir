// GROUPE D’ENTRÉE — CRÉATION DU SOCLE FRANÇAIS
//    Ce groupe crée le Bundle final de l’étape 3 avec Composition et Patient.
//    Les autres ressources administratives sont créées dans les groupes CDA qui
//    les justifient, afin d’éviter les ressources optionnelles vides
group TransformClinicalDocumentToFrenchFHIRDocument(source cda : ClinicalDocument, target bundle : Bundle) {
  cda ->  bundle.entry as compositionEntry,  compositionEntry.resource = create('Composition') as composition,  composition.id = uuid(),  compositionEntry.fullUrl = ('urn:uuid:' + %composition.id),  bundle.entry as patientEntry,  patientEntry.resource = create('Patient') as patient,  patient.id = uuid(),  patientEntry.fullUrl = ('urn:uuid:' + %patient.id) then OrchestrateClinicalDocumentFrenchTransformation(cda, patient, composition, bundle) "main";
}

// GROUPE D’ORCHESTRATION — SOCLE GÉNÉRIQUE + ENRICHISSEMENT FRANÇAIS
//    Ce groupe suit la structure du ClinicalDocument CDA.
//    Il appelle d’abord le traitement français de l’en-tête, où chaque ressource
//    optionnelle est créée dans le groupe correspondant au bloc CDA. Il appelle
//    ensuite le corps générique de l’étape 2 pour créer les sections narratives
group OrchestrateClinicalDocumentFrenchTransformation(source cda, target patient, target composition, target bundle) {
  // CDA ClinicalDocument.header -> socle générique + enrichissement français
  cda then MapClinicalDocumentHeaderWithFrenchRules(cda, bundle, composition, patient) "headerFrenchBase";
  // CDA ClinicalDocument.structuredBody -> FHIR Composition.section title/code/text
  cda then MapClinicalDocumentBody(cda, composition, bundle, patient) "body";
}

// GROUPE — EN-TÊTE CDA AVEC RÈGLES FRANÇAISES
//    Ce groupe reprend l’organisation de l’en-tête CDA. Les ressources optionnelles
//    sont créées dans les sous-groupes correspondant aux blocs CDA, puis remplies
//    avec les groupes génériques de l’étape 2 et enrichies avec les groupes français
group MapClinicalDocumentHeaderWithFrenchRules(source cda, target bundle, target composition, target patient) {
  // CDA ClinicalDocument -> profils généraux Bundle / Composition
  // cda -> bundle.meta as bundleMeta then { cda -> bundleMeta.profile = 'http://hl7.org/fhir/StructureDefinition/Bundle' "bundleGenericProfile"; } "bundleMetaProfile";
  // cda -> composition.meta as compositionMeta then { cda -> compositionMeta.profile = 'http://hl7.org/fhir/StructureDefinition/Composition' "compositionGenericProfile"; } "compositionMetaProfile";
  // CDA ClinicalDocument -> FHIR Bundle.meta.profile EU EPS
  cda -> bundle.meta as bundleMeta then {
    cda -> bundleMeta.profile = 'http://hl7.eu/fhir/eps/StructureDefinition/bundle-eu-eps' "bundleEUEPSProfile";
  } "bundleMetaProfile";
  // CDA ClinicalDocument -> FHIR Composition.meta.profile EU EPS
  cda -> composition.meta as compositionMeta then {
    cda -> compositionMeta.profile = 'http://hl7.eu/fhir/eps/StructureDefinition/composition-eu-eps' "compositionEUEPSProfile";
  } "compositionMetaProfile";
  // CDA id/setId/versionNumber/effectiveTime -> FHIR métadonnées Bundle + Composition
  cda then MapHeaderDocumentIdentity(cda, bundle, composition) "documentIdentity";
  // CDA languageCode/code/title/effectiveTime/confidentialityCode -> FHIR description de la Composition
  cda then MapHeaderDocumentDescription(cda, composition, patient) "documentDescription";
  // CDA recordTarget.patientRole -> Patient générique + enrichissement français
  cda then MapHeaderRecordTarget(cda, patient, composition) "recordTargetBase";
  cda.recordTarget as recordTarget then {
    recordTarget.patientRole as patientRole then MapCDARecordTargetToFrenchPatient(patientRole, patient) "patientFR";
  } "recordTargetFR";
  // CDA custodian -> Organization custodian générique + française
  cda then MapHeaderCustodianWithFrenchRules(cda, composition, bundle) "custodian";
  // CDA author -> Practitioner / Organization / PractitionerRole génériques + français
  cda then MapHeaderAuthorWithFrenchRules(cda, composition, bundle) "author";
  // CDA legalAuthenticator -> signataire légal générique + français
  cda then MapHeaderLegalAuthenticatorWithFrenchRules(cda, composition, bundle) "legalAuthenticator";
  // CDA documentationOf.serviceEvent -> FHIR Composition.event
  cda then MapHeaderDocumentationOf(cda, composition) "documentationOf";
  // CDA componentOf.encompassingEncounter -> Encounter / Location / Organization génériques + français
  cda then MapHeaderEncounterContextWithFrenchRules(cda, composition, bundle, patient) "encounterContext";
  // CDA informant ECON/NOK -> RelatedPerson génériques + françaises
  cda then MapHeaderEmergencyContactWithFrenchRules(cda, bundle, patient) "emergencyContact";
  cda then MapHeaderTrustedPersonWithFrenchRules(cda, bundle, patient) "trustedPerson";
  // CDA guardian -> RelatedPerson générique + française
  cda then MapHeaderGuardianWithFrenchRules(cda, bundle, patient) "guardian";
  // CDA relatedDocument -> FHIR Composition.relatesTo
  cda then MapHeaderRelatedDocument(cda, composition) "relatedDocument";
}

// CDA custodian -> création Organization + socle générique + enrichissement français
group MapHeaderCustodianWithFrenchRules(source src, target composition, target bundle) {
  src.custodian as custodian then {
    custodian.assignedCustodian as assignedCustodian then {
      assignedCustodian.representedCustodianOrganization as representedOrg ->  bundle.entry as custodianOrgEntry,  custodianOrgEntry.resource = create('Organization') as custodianOrganization,  custodianOrganization.id = uuid(),  custodianOrgEntry.fullUrl = ('urn:uuid:' + %custodianOrganization.id) then {
        representedOrg then MapCDAOrganizationToOrganization(representedOrg, custodianOrganization) "custodianOrganizationBase";
        representedOrg then MapCDAOrganizationToFrenchOrganization(representedOrg, custodianOrganization) "custodianOrganizationFR";
        representedOrg ->  composition.custodian = create('Reference') as custodianRef,  custodianRef.reference = ('urn:uuid:' + %custodianOrganization.id) "custodianRef";
      } "custodianOrganizationResource";
    };
  };
}

// CDA author -> création Practitioner / Organization / PractitionerRole + enrichissement français
group MapHeaderAuthorWithFrenchRules(source src, target composition, target bundle) {
  src.author as author then {
    author.assignedAuthor as assignedAuthor ->  bundle.entry as practitionerEntry,  practitionerEntry.resource = create('Practitioner') as authorPractitioner,  authorPractitioner.id = uuid(),  practitionerEntry.fullUrl = ('urn:uuid:' + %authorPractitioner.id) then {
      assignedAuthor then MapCDAAssignedEntityToPractitioner(assignedAuthor, authorPractitioner) "authorPractitionerBase";
      assignedAuthor then MapCDAAssignedEntityToFrenchPractitioner(assignedAuthor, authorPractitioner) "authorPractitionerFR";
      assignedAuthor.representedOrganization as representedOrganization ->  bundle.entry as organizationEntry,  organizationEntry.resource = create('Organization') as authorOrganization,  authorOrganization.id = uuid(),  organizationEntry.fullUrl = ('urn:uuid:' + %authorOrganization.id),  bundle.entry as roleEntry,  roleEntry.resource = create('PractitionerRole') as authorRole,  authorRole.id = uuid(),  roleEntry.fullUrl = ('urn:uuid:' + %authorRole.id) then {
        representedOrganization then MapCDAOrganizationToOrganization(representedOrganization, authorOrganization) "authorOrganizationBase";
        representedOrganization then MapCDAOrganizationToFrenchOrganization(representedOrganization, authorOrganization) "authorOrganizationFR";
        assignedAuthor then MapCDAAssignedEntityToPractitionerRole(assignedAuthor, authorRole, authorPractitioner, authorOrganization) "authorRoleBase";
        assignedAuthor then MapCDAAssignedEntityToFrenchPractitionerRole(assignedAuthor, authorRole, authorPractitioner, authorOrganization) "authorRoleFR";
        assignedAuthor ->  composition.author as authorRef,  authorRef.reference = ('urn:uuid:' + %authorRole.id) "compositionAuthorRoleRef";
      } "authorWithOrganization";
      assignedAuthor where (assignedAuthor.representedOrganization.empty()) ->  composition.author as authorRef,  authorRef.reference = ('urn:uuid:' + %authorPractitioner.id) "compositionAuthorPractitionerRef";
    } "authorPractitionerResource";
  };
}

// CDA legalAuthenticator -> création Practitioner / Organization / PractitionerRole + enrichissement français
group MapHeaderLegalAuthenticatorWithFrenchRules(source src, target composition, target bundle) {
  src.legalAuthenticator as legalAuthenticator then {
    legalAuthenticator.assignedEntity as assignedEntity ->  bundle.entry as practitionerEntry,  practitionerEntry.resource = create('Practitioner') as legalPractitioner,  legalPractitioner.id = uuid(),  practitionerEntry.fullUrl = ('urn:uuid:' + %legalPractitioner.id) then {
      assignedEntity then MapCDAAssignedEntityToPractitioner(assignedEntity, legalPractitioner) "legalPractitionerBase";
      assignedEntity then MapCDAAssignedEntityToFrenchPractitioner(assignedEntity, legalPractitioner) "legalPractitionerFR";
      legalAuthenticator -> composition.attester as attester then {
        legalAuthenticator -> attester.mode = 'legal' "mode";
        legalAuthenticator.time as time -> attester.time = create('dateTime') as dt then TSDateTime(time, dt);
        assignedEntity.representedOrganization as representedOrganization ->  bundle.entry as organizationEntry,  organizationEntry.resource = create('Organization') as legalOrganization,  legalOrganization.id = uuid(),  organizationEntry.fullUrl = ('urn:uuid:' + %legalOrganization.id),  bundle.entry as roleEntry,  roleEntry.resource = create('PractitionerRole') as legalRole,  legalRole.id = uuid(),  roleEntry.fullUrl = ('urn:uuid:' + %legalRole.id) then {
          representedOrganization then MapCDAOrganizationToOrganization(representedOrganization, legalOrganization) "legalOrganizationBase";
          representedOrganization then MapCDAOrganizationToFrenchOrganization(representedOrganization, legalOrganization) "legalOrganizationFR";
          assignedEntity then MapCDAAssignedEntityToPractitionerRole(assignedEntity, legalRole, legalPractitioner, legalOrganization) "legalRoleBase";
          assignedEntity then MapCDAAssignedEntityToFrenchPractitionerRole(assignedEntity, legalRole, legalPractitioner, legalOrganization) "legalRoleFR";
          assignedEntity ->  attester.party = create('Reference') as partyRef,  partyRef.reference = ('urn:uuid:' + %legalRole.id) "partyRole";
        } "legalWithOrganization";
        assignedEntity where (assignedEntity.representedOrganization.empty()) ->  attester.party = create('Reference') as partyRef,  partyRef.reference = ('urn:uuid:' + %legalPractitioner.id) "partyPractitioner";
      } "attester";
    } "legalPractitionerResource";
  };
}

// CDA componentOf.encompassingEncounter -> création Encounter / Location / Organization + enrichissement français
group MapHeaderEncounterContextWithFrenchRules(source src, target composition, target bundle, target patient) {
  src.componentOf as componentOf then {
    componentOf.encompassingEncounter as encompassingEncounter ->  bundle.entry as encounterEntry,  encounterEntry.resource = create('Encounter') as encounter,  encounter.id = uuid(),  encounterEntry.fullUrl = ('urn:uuid:' + %encounter.id) then {
      // CDA encompassingEncounter -> FHIR Composition.encounter
      encompassingEncounter ->  composition.encounter = create('Reference') as encRef,  encRef.reference = ('urn:uuid:' + %encounter.id) "compositionEncounterRef";
      // Cas 1 : healthCareFacility présent -> Encounter + Location + Organization
      encompassingEncounter.location as srcLocation then {
        srcLocation.healthCareFacility as facility ->  bundle.entry as locationEntry,  locationEntry.resource = create('Location') as location,  location.id = uuid(),  locationEntry.fullUrl = ('urn:uuid:' + %location.id),  bundle.entry as facilityOrganizationEntry,  facilityOrganizationEntry.resource = create('Organization') as facilityOrganization,  facilityOrganization.id = uuid(),  facilityOrganizationEntry.fullUrl = ('urn:uuid:' + %facilityOrganization.id) then {
          // Groupe français d’orchestration :
          // appelle le générique de l’étape 2 puis les enrichissements français de l’étape 3.
          encompassingEncounter then MapCDAEncounterContextToFrenchResources(encompassingEncounter, facility, encounter, patient, location, facilityOrganization) "encounterContextFR";
        } "facilityResources";
      };
      // Cas 2 : encompassingEncounter présent mais sans location
      encompassingEncounter where (encompassingEncounter.location.empty()) then MapCDAEncounterOnlyToFrenchResources(encompassingEncounter, encounter, patient) "encounterOnlyFR";
      // Cas 3 : location présente mais sans healthCareFacility
      encompassingEncounter.location as srcLocation where (srcLocation.healthCareFacility.empty()) then MapCDAEncounterOnlyToFrenchResources(encompassingEncounter, encounter, patient) "encounterOnlyFRWithoutFacility";
    } "encounterResource";
  };
}

      relatedEntity then MapCDARelatedEntityToRelatedPerson(relatedEntity, emergencyContactRelatedPerson, patient) "relatedPersonBase";
      relatedEntity then MapCDAInformantToFrenchRelatedPerson(relatedEntity, emergencyContactRelatedPerson, patient) "relatedPersonFR";

  };
}

// CDA guardian -> création RelatedPerson + enrichissement français
group MapHeaderGuardianWithFrenchRules(source src, target bundle, target patient) {
  src.recordTarget as recordTarget then {
    recordTarget.patientRole as patientRole then {
      patientRole.patient as patientCda then {
        patientCda.guardian as guardian ->  bundle.entry as guardianEntry,  guardianEntry.resource = create('RelatedPerson') as guardianRelatedPerson,  guardianRelatedPerson.id = uuid(),  guardianEntry.fullUrl = ('urn:uuid:' + %guardianRelatedPerson.id) then {
          guardian ->  guardianRelatedPerson.patient = create('Reference') as patientRef,  patientRef.reference = ('urn:uuid:' + %patient.id) "patientRef";
          guardian.id as id -> guardianRelatedPerson.identifier = create('Identifier') as identifier then II(id, identifier) "identifier";
          guardian.addr as addr -> guardianRelatedPerson.address = create('Address') as address then ADAddress(addr, address) "address";
          guardian.telecom as telecom -> guardianRelatedPerson.telecom = create('ContactPoint') as contactPoint then TELContactPoint(telecom, contactPoint);
          guardian.guardianPerson as guardianPerson then {
            guardianPerson.name as name -> guardianRelatedPerson.name = create('HumanName') as humanName then ENHumanName(name, humanName);
          };
          guardian then MapCDAGuardianToFrenchRelatedPerson(guardian, guardianRelatedPerson, patient) "guardianFR";
        } "guardianResource";
      };
    };
  };
}

group MapCDARecordTargetToFrenchPatient(source src, target tgt) {
  // CDA patientRole -> FHIR Patient.meta.profile
  src -> tgt.meta as meta then {
    src -> meta.profile = 'https://hl7.fr/ig/fhir/core/StructureDefinition/fr-core-patient-ins' "patientINSProfile";
  } "meta";
  src.patient as patient then {
    // CDA patient.name -> FHIR Patient.name officiel
    patient.name as patientName -> tgt.name = create('HumanName') as officialName then MapCDAPatientNameToFrenchOfficialName(patientName, officialName) "officialName";
    // CDA patient.birthplace -> FHIR Patient.extension birthPlace
    patient.birthplace as birthplace -> tgt.extension = create('Extension') as birthPlaceExt then MapCDABirthPlaceToFHIRPatientExtension(birthplace, birthPlaceExt) "birthPlace";
  } "patientDataFR";
  // CDA patientRole.id -> FHIR Patient.identifier INS
  src.id as id then MapCDAIIToPatientINSIdentifiers(id, tgt) "insIdentifiers";
  // CDA patientRole.id -> FHIR Patient.identifier PI hors INS
  src.id as id then MapCDAIIToPatientOtherIdentifiers(id, tgt) "otherIdentifiers";
  // CDA patientRole.id INS-NIR -> FHIR Patient.extension identity reliability VALI
  src.id as id where (id.root = '1.2.250.1.213.1.4.8') -> tgt.extension = create('Extension') as identityReliability then MapCDAIIToIdentityReliabilityVALI(id, identityReliability) "identityReliabilityNIR";
  // CDA patientRole.id INS-NIA -> FHIR Patient.extension identity reliability VALI
  src.id as id where (id.root = '1.2.250.1.213.1.4.9') -> tgt.extension = create('Extension') as identityReliability then MapCDAIIToIdentityReliabilityVALI(id, identityReliability) "identityReliabilityNIA";
  // CDA patientRole.id INS-NIR-TEST -> FHIR Patient.extension identity reliability VALI
  src.id as id where (id.root = '1.2.250.1.213.1.4.10') -> tgt.extension = create('Extension') as identityReliability then MapCDAIIToIdentityReliabilityVALI(id, identityReliability) "identityReliabilityNIRTest";
  // CDA patientRole.id INS-NIR-DEMO -> FHIR Patient.extension identity reliability VALI
  src.id as id where (id.root = '1.2.250.1.213.1.4.11') -> tgt.extension = create('Extension') as identityReliability then MapCDAIIToIdentityReliabilityVALI(id, identityReliability) "identityReliabilityNIRDemo";
}

// CDA PN patient.name -> FHIR HumanName officiel
group MapCDAPatientNameToFrenchOfficialName(source src, target tgt : HumanName) {
  // CDA patient.name -> FHIR HumanName.use
  src -> tgt.use = 'official' "use";
  src.item as item then {
    // CDA family qualifier=BR -> FHIR HumanName.family
    item.family as family where (family.qualifier = 'BR') -> tgt.family = (family.xmlText);
    // CDA given qualifier=BR -> FHIR HumanName.given
    item.given as given where (given.qualifier = 'BR') -> tgt.given = (given.xmlText);
    // CDA given sans qualifier -> FHIR HumanName.extension liste des prénoms de naissance
    item.given as givenList where (givenList.qualifier.empty()) -> tgt.extension = create('Extension') as ext then MapCDAGivenNameToBirthListGivenNameExtension(givenList, ext) "birthListGivenName";
  } "items";
}

// CDA given sans qualifier -> FHIR extension liste des prénoms de naissance
group MapCDAGivenNameToBirthListGivenNameExtension(source src, target tgt : Extension) {
  // CDA given -> FHIR Extension.url
  src -> tgt.url = 'https://hl7.fr/ig/fhir/core/StructureDefinition/fr-core-patient-birth-list-given-name' "url";
  // CDA given.xmlText -> FHIR Extension.valueString
  src.xmlText as txt -> tgt.value = cast(txt, 'string') "valueString";
}

// CDA patient.birthplace -> FHIR Patient.extension birthPlace
group MapCDABirthPlaceToFHIRPatientExtension(source src, target tgt : Extension) {
  // CDA birthplace -> FHIR Extension.url
  src -> tgt.url = 'http://hl7.org/fhir/StructureDefinition/patient-birthPlace' "url";
  // CDA birthplace.place.addr -> FHIR Extension.valueAddress
  src.place as place -> tgt.value = create('Address') as addr then {
    place.addr as placeAddr then ADAddress(placeAddr, addr) "baseAddress";
    place.addr as placeAddr then {
      placeAddr.item as item then {
        // CDA county -> FHIR Address.extension code INSEE
        item.county as insee -> addr.extension = create('Extension') as inseeExt then {
          insee -> inseeExt.url = 'https://hl7.fr/ig/fhir/core/StructureDefinition/fr-core-address-insee-code' "url";
          insee.xmlText as txt -> inseeExt.value = create('Coding') as coding then {
            txt -> coding.code = cast(txt, 'code') "code";
          } "valueCoding";
        } "inseeCodeExtension";
      };
    } "county";
  } "valueAddress";
}

// IDENTIFIANTS PATIENT FR
// CDA II patientRole.id -> FHIR Patient.identifier INS
group MapCDAIIToPatientINSIdentifiers(source src, target tgt) {
  // CDA root INS-NIR -> FHIR Identifier INS-NIR
  src where (src.root = '1.2.250.1.213.1.4.8') -> tgt.identifier = create('Identifier') as id then MapCDAIIToINSNIRIdentifier(src, id) "INSNIR";
  // CDA root INS-NIA -> FHIR Identifier INS-NIA
  src where (src.root = '1.2.250.1.213.1.4.9') -> tgt.identifier = create('Identifier') as id then MapCDAIIToINSNIAIdentifier(src, id) "INSNIA";
  // CDA root INS-NIR-TEST -> FHIR Identifier INS-NIR-TEST
  src where (src.root = '1.2.250.1.213.1.4.10') -> tgt.identifier = create('Identifier') as id then MapCDAIIToINSNIRTestIdentifier(src, id) "INSNIRTEST";
  // CDA root INS-NIR-DEMO -> FHIR Identifier INS-NIR-DEMO
  src where (src.root = '1.2.250.1.213.1.4.11') -> tgt.identifier = create('Identifier') as id then MapCDAIIToINSNIRDemoIdentifier(src, id) "INSNIRDEMO";
}

// CDA II INS-NIR -> FHIR Identifier INS-NIR
group MapCDAIIToINSNIRIdentifier(source src, target tgt : Identifier) {
  src -> tgt then II(src, tgt) "base";
  src -> tgt.use = 'official' "use";
  src -> tgt.type = create('CodeableConcept') as idType then {
    src -> idType.coding = create('Coding') as idCoding then {
      src -> idCoding.system = 'https://hl7.fr/ig/fhir/core/CodeSystem/fr-core-cs-v2-0203' "system";
      src -> idCoding.code = 'INS-NIR' "code";
    } "coding";
  } "type";
}

// CDA healthCareFacility -> FHIR Organization facility française
group MapCDAHealthCareFacilityToFrenchOrganization(source src, target tgt) {
  // Cas principal : healthCareFacility.serviceProviderOrganization existe.
  // On réutilise le groupe français général des Organization pour éviter de
  // reprofiler manuellement ici.
  src.serviceProviderOrganization as serviceProviderOrganization then MapCDAOrganizationToFrenchOrganization(serviceProviderOrganization, tgt) "serviceProviderOrganizationFR";
  // Cas complémentaire : si le healthCareFacility porte lui-même un identifiant,
  // on l’ajoute aussi à l’Organization de prise en charge.
  src.id as id then MapCDAIIToFrenchOrganizationIdentifiers(id, tgt) "facilityIdentifiers";
  // Cas de secours : si aucune serviceProviderOrganization n’existe, on applique
  // quand même les profils/identifiants français via le groupe général.
  src where (src.serviceProviderOrganization.empty()) then MapCDAOrganizationToFrenchOrganization(src, tgt) "facilityFR";
  // CDA healthCareFacility sans id ni organization -> FHIR Organization.name data-absent-reason
  src where (src.id.empty() and src.serviceProviderOrganization.empty()) -> tgt.name = create('string') as nameString then {
    src -> nameString.extension = create('Extension') as absentReason then {
      src -> absentReason.url = 'http://hl7.org/fhir/StructureDefinition/data-absent-reason' "url";
      src -> absentReason.value = cast('unknown', 'code') "valueCode";
    } "dataAbsentReason";
  } "facilityOrganizationNameAbsentReason";
  // CDA serviceProviderOrganization sans nom ni id -> FHIR Organization.name data-absent-reason
  src.serviceProviderOrganization as serviceProviderOrganization where (serviceProviderOrganization.name.empty() and serviceProviderOrganization.id.empty()) -> tgt.name = create('string') as nameString then {
    serviceProviderOrganization -> nameString.extension = create('Extension') as absentReason then {
      serviceProviderOrganization -> absentReason.url = 'http://hl7.org/fhir/StructureDefinition/data-absent-reason' "url";
      serviceProviderOrganization -> absentReason.value = cast('unknown', 'code') "valueCode";
    } "dataAbsentReason";
  } "serviceProviderOrganizationNameAbsentReason";
}

// ENCOUNTER / LOCATION / FACILITY FR
// CDA encompassingEncounter + healthCareFacility -> FHIR Encounter / Location / Organization génériques + français
group MapCDAEncounterContextToFrenchResources(source encompassingEncounter, source facility, target encounter, target patient, target location, target facilityOrganization) {
  // CDA encompassingEncounter -> FHIR Encounter générique étape 2
  encompassingEncounter then MapCDAEncompassingEncounterToEncounter(encompassingEncounter, encounter, patient) "encounterBase";
  // CDA encompassingEncounter -> FHIR Encounter enrichi français étape 3
  encompassingEncounter then MapCDAEncompassingEncounterToFrenchEncounter(encompassingEncounter, encounter, patient) "encounterFR";
  // CDA healthCareFacility -> FHIR Location / Organization génériques étape 2
  facility then MapCDAHealthCareFacilityToLocationContext(facility, encounter, location, facilityOrganization) "facilityBase";
  // CDA healthCareFacility -> FHIR Location française étape 3
  facility then MapCDAHealthCareFacilityToFrenchLocation(facility, location, facilityOrganization) "locationFR";
  // CDA healthCareFacility -> FHIR Organization française étape 3
  facility then MapCDAHealthCareFacilityToFrenchOrganization(facility, facilityOrganization) "facilityOrganizationFR";
}

group MapCDAEncompassingEncounterToFrenchEncounter(source src, target tgt, target patient) {
  // CDA encompassingEncounter -> FHIR Encounter.meta.profile
  src -> tgt.meta as metaBlock then {
    src -> metaBlock.profile = 'https://hl7.fr/ig/fhir/core/StructureDefinition/fr-core-encounter' "profile";
  } "meta";
  // CDA encompassingEncounter.code -> FHIR Encounter.type NOS
  src.code as srcCode -> tgt.type = create('CodeableConcept') as cc then {
    srcCode.code as code -> cc.coding = create('Coding') as coding then {
      code -> coding.system = 'https://mos.esante.gouv.fr/NOS/TRE_R304-HL7v3ActCode/FHIR/TRE-R304-HL7v3ActCode' "typeSystem";
      code -> coding.code = cast(code, 'code') "typeCode";
      srcCode.displayName as display -> coding.display = cast(display, 'string') "typeDisplay";
    } "coding";
  } "encounterType";
}

// CDA healthCareFacility -> FHIR Location française
group MapCDAHealthCareFacilityToFrenchLocation(source src, target tgt, target facilityOrganization) {
  // CDA healthCareFacility -> FHIR Location.meta.profile
  src -> tgt.meta as metaBlock then {
    src -> metaBlock.profile = 'https://hl7.fr/ig/fhir/core/StructureDefinition/fr-core-location' "frCoreLocationProfile";
    src -> metaBlock.profile = 'https://interop.esante.gouv.fr/ig/fhir/annuaire/StructureDefinition/as-location' "asLocationProfile";
  } "meta";
  // CDA healthCareFacility.location.name -> FHIR Location.name
  src.location as facilityLocation then {
    facilityLocation.name as locName then {
      locName.item as item -> tgt.name = (item.xmlText) "locationName";
    } "locationName";
  } "facilityLocation";
  // CDA healthCareFacility sans location -> FHIR Location.name data-absent-reason
  src where (src.location.empty()) -> tgt.name = create('string') as nameString then {
    src -> nameString.extension = create('Extension') as absentReason then {
      src -> absentReason.url = 'http://hl7.org/fhir/StructureDefinition/data-absent-reason' "url";
      src -> absentReason.value = cast('unknown', 'code') "valueCode";
    } "dataAbsentReason";
  } "locationNameAbsentReasonNoLocation";
  // CDA healthCareFacility.location sans name -> FHIR Location.name data-absent-reason
  src.location as facilityLocation where (facilityLocation.name.empty()) -> tgt.name = create('string') as nameString then {
    facilityLocation -> nameString.extension = create('Extension') as absentReason then {

  } "locationNameAbsentReasonNoName";
  // CDA healthCareFacility.code -> FHIR Location.type NOS
  src.code as srcCode -> tgt.type = create('CodeableConcept') as cc then {
    srcCode.code as code -> cc.coding = create('Coding') as coding then {
