<span style="color: navy">// </span><span style="color: green">   Ce groupe traite l’en-tête du ClinicalDocument CDA bloc par bloc.</span>

<span style="color: navy">// </span><span style="color: green">   Pour chaque élément de l’en-tête CDA, il appelle le groupe spécialisé</span>
<span style="color: navy">// </span><span style="color: green">   correspondant :</span>
<span style="color: navy">// </span><span style="color: green">   - id, setId, versionNumber et effectiveTime sont transmis à</span>
<span style="color: navy">// </span><span style="color: green">     MapHeaderDocumentIdentity pour alimenter les métadonnées du Bundle et de</span>
<span style="color: navy">// </span><span style="color: green">     la Composition ;</span>
<span style="color: navy">// </span><span style="color: green">   - languageCode, code, title, effectiveTime et confidentialityCode sont transmis</span>
<span style="color: navy">// </span><span style="color: green">     à MapHeaderDocumentDescription pour alimenter la description documentaire ;</span>
<span style="color: navy">// </span><span style="color: green">   - recordTarget.patientRole est transmis à MapHeaderRecordTarget pour alimenter</span>
<span style="color: navy">// </span><span style="color: green">     le Patient et la référence Composition.subject ;</span>
<span style="color: navy">// </span><span style="color: green">   - custodian est transmis à MapHeaderCustodian pour créer et alimenter</span>
<span style="color: navy">// </span><span style="color: green">     l’Organization responsable de la conservation du document ;</span>
<span style="color: navy">// </span><span style="color: green">   - author est transmis à MapHeaderAuthor pour créer et alimenter le Practitioner,</span>

<span style="color: navy">// </span><span style="color: green">   - legalAuthenticator est transmis à MapHeaderLegalAuthenticator pour créer et</span>

<span style="color: navy">// </span><span style="color: green">   - documentationOf.serviceEvent est transmis à MapHeaderDocumentationOf pour</span>
<span style="color: navy">// </span><span style="color: green">     alimenter Composition.event ;</span>
<span style="color: navy">// </span><span style="color: green">   - componentOf.encompassingEncounter est transmis à MapHeaderEncounterContext</span>
<span style="color: navy">// </span><span style="color: green">     pour créer et alimenter Encounter, Location et Organization si le bloc CDA</span>
<span style="color: navy">// </span><span style="color: green">     est présent ;</span>
<span style="color: navy">// </span><span style="color: green">   - informant.relatedEntity classCode=ECON est transmis à MapHeaderEmergencyContact</span>
<span style="color: navy">// </span><span style="color: green">     pour créer la RelatedPerson correspondant à la personne à prévenir ;</span>
<span style="color: navy">// </span><span style="color: green">   - informant.relatedEntity classCode=NOK est transmis à MapHeaderTrustedPerson</span>
<span style="color: navy">// </span><span style="color: green">     pour créer la RelatedPerson correspondant à la personne de confiance ;</span>
<span style="color: navy">// </span><span style="color: green">   - recordTarget.patientRole.patient.guardian est transmis à MapHeaderGuardian</span>
<span style="color: navy">// </span><span style="color: green">     pour créer la RelatedPerson correspondant au guardian ;</span>
<span style="color: navy">// </span><span style="color: green">   - relatedDocument est transmis à MapHeaderRelatedDocument pour alimenter</span>
<span style="color: navy">// </span><span style="color: green">     Composition.relatesTo.</span>

<span style="color: navy">// </span><span style="color: green">   Les ressources optionnelles sont créées dans le groupe qui traite le bloc CDA</span>
<span style="color: navy">// </span><span style="color: green">   correspondant. Ainsi, une ressource FHIR n’est créée que lorsqu’un élément CDA</span>
<span style="color: navy">// </span><span style="color: green">   justifie son existence</span>
<b>group </b>MapClinicalDocumentHeader<span style="color: navy">(</span><b>source</b> <span style="color: maroon">cda</span><span style="color: navy"> : </span>ClinicalDocument, <b>target</b> <span style="color: maroon">bundle</span><span style="color: navy"> : </span>Bundle, <b>target</b> <span style="color: maroon">composition</span><span style="color: navy"> : </span>Composition, <b>target</b> <span style="color: maroon">patient</span><span style="color: navy"> : </span>Patient<span style="color: navy">)</span><span style="color: navy"> {

  cda<b> then </b>MapHeaderDocumentIdentity<span style="color: navy">(</span><span style="color: maroon">cda</span><span style="color: navy">, </span><span style="color: maroon">bundle</span><span style="color: navy">, </span><span style="color: maroon">composition</span><span style="color: navy">)</span> <i>&quot;documentIdentity&quot;</i><span style="color: navy">;</span>
  <span style="color: navy">// </span><span style="color: green">CDA languageCode/code/title/effectiveTime/confidentialityCode -&gt; FHIR description de la Composition</span>

  <span style="color: navy">// </span><span style="color: green">CDA recordTarget.patientRole -&gt; FHIR Patient + Composition.subject</span>
